# React 컴포넌트와 REST API 연동

React 컴포넌트에서 REST API를 연동하는 방법을 단계별로 설명드리겠습니다.

## 1. 기본 fetch 사용

가장 기본적인 방법으로 `fetch` API를 사용합니다:

```javascript
import { useState, useEffect } from "react";

function UserList() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    fetch("https://api.example.com/users")
      .then((response) => {
        if (!response.ok) {
          throw new Error("네트워크 응답이 올바르지 않습니다");
        }
        return response.json();
      })
      .then((data) => {
        setUsers(data);
        setLoading(false);
      })
      .catch((error) => {
        setError(error.message);
        setLoading(false);
      });
  }, []);

  if (loading) return <div>로딩 중...</div>;
  if (error) return <div>에러: {error}</div>;

  return (
    <ul>
      {users.map((user) => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}
```

## 2. async/await 패턴

더 읽기 쉬운 코드를 위해 async/await를 사용할 수 있습니다:

```javascript
import { useState, useEffect } from "react";

function UserList() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const fetchUsers = async () => {
      try {
        const response = await fetch("https://api.example.com/users");
        if (!response.ok) {
          throw new Error("데이터를 가져오는데 실패했습니다");
        }
        const data = await response.json();
        setUsers(data);
      } catch (error) {
        setError(error.message);
      } finally {
        setLoading(false);
      }
    };

    fetchUsers();
  }, []);

  if (loading) return <div>로딩 중...</div>;
  if (error) return <div>에러: {error}</div>;

  return (
    <ul>
      {users.map((user) => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}
```

## 3. CRUD 작업 전체 예제

생성, 조회, 수정, 삭제 기능을 모두 포함한 예제입니다:

```javascript
import { useState, useEffect } from "react";

function TodoApp() {
  const [todos, setTodos] = useState([]);
  const [newTodo, setNewTodo] = useState("");
  const [loading, setLoading] = useState(true);

  const API_URL = "https://api.example.com/todos";

  // GET - 목록 조회
  useEffect(() => {
    fetchTodos();
  }, []);

  const fetchTodos = async () => {
    try {
      const response = await fetch(API_URL);
      const data = await response.json();
      setTodos(data);
    } catch (error) {
      console.error("조회 실패:", error);
    } finally {
      setLoading(false);
    }
  };

  // POST - 새로운 항목 생성
  const createTodo = async (e) => {
    e.preventDefault();
    try {
      const response = await fetch(API_URL, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ title: newTodo, completed: false }),
      });
      const data = await response.json();
      setTodos([...todos, data]);
      setNewTodo("");
    } catch (error) {
      console.error("생성 실패:", error);
    }
  };

  // PUT - 항목 수정
  const updateTodo = async (id, completed) => {
    try {
      const response = await fetch(`${API_URL}/${id}`, {
        method: "PUT",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ completed }),
      });
      const updatedTodo = await response.json();
      setTodos(todos.map((todo) => (todo.id === id ? updatedTodo : todo)));
    } catch (error) {
      console.error("수정 실패:", error);
    }
  };

  // DELETE - 항목 삭제
  const deleteTodo = async (id) => {
    try {
      await fetch(`${API_URL}/${id}`, {
        method: "DELETE",
      });
      setTodos(todos.filter((todo) => todo.id !== id));
    } catch (error) {
      console.error("삭제 실패:", error);
    }
  };

  if (loading) return <div>로딩 중...</div>;

  return (
    <div>
      <form onSubmit={createTodo}>
        <input
          value={newTodo}
          onChange={(e) => setNewTodo(e.target.value)}
          placeholder="할 일 입력"
        />
        <button type="submit">추가</button>
      </form>

      <ul>
        {todos.map((todo) => (
          <li key={todo.id}>
            <input
              type="checkbox"
              checked={todo.completed}
              onChange={(e) => updateTodo(todo.id, e.target.checked)}
            />
            {todo.title}
            <button onClick={() => deleteTodo(todo.id)}>삭제</button>
          </li>
        ))}
      </ul>
    </div>
  );
}
```

## 5. 커스텀 Hook으로 재사용성 높이기

```javascript
import { useState, useEffect } from "react";

function useFetch(url) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const fetchData = async () => {
      try {
        const response = await fetch(url);
        if (!response.ok) throw new Error("데이터 조회 실패");
        const result = await response.json();
        setData(result);
      } catch (error) {
        setError(error.message);
      } finally {
        setLoading(false);
      }
    };

    fetchData();
  }, [url]);

  return { data, loading, error };
}

// 사용 예시
function UserList() {
  const {
    data: users,
    loading,
    error,
  } = useFetch("https://api.example.com/users");

  if (loading) return <div>로딩 중...</div>;
  if (error) return <div>에러: {error}</div>;

  return (
    <ul>
      {users.map((user) => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}
```

# React 컴포넌트와 REST API 연동 - 개념 설명

React에서 REST API를 연동한다는 것은 **서버의 데이터를 가져와서 화면에 표시하고, 사용자의 액션을 서버에 전달하는 과정**입니다. 핵심 개념들을 차근차근 설명드리겠습니다.

## 1. REST API란?

**REST API**는 HTTP 프로토콜을 사용해 서버와 통신하는 방식입니다.

### HTTP 메서드와 CRUD 작업

```
GET    → 조회 (Read)     예: 사용자 목록 가져오기
POST   → 생성 (Create)   예: 새 사용자 등록
PUT    → 수정 (Update)   예: 사용자 정보 변경
DELETE → 삭제 (Delete)   예: 사용자 삭제
```

### API 요청 예시

```javascript
// GET 요청
fetch("https://api.example.com/users");

// POST 요청
fetch("https://api.example.com/users", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ name: "홍길동" }),
});
```

## 2. React에서 API를 연동하는 이유

React는 **UI 라이브러리**로, 데이터는 별도로 관리됩니다.

```
┌─────────────┐      API 요청      ┌─────────────┐
│   React     │ ─────────────────→ │   서버      │
│ (프론트엔드) │                     │ (백엔드)    │
│             │ ←───────────────── │             │
└─────────────┘      데이터 응답    └─────────────┘
```

**분리하는 이유:**

- 프론트엔드와 백엔드를 독립적으로 개발 가능
- 같은 API를 웹, 모바일 앱에서 재사용
- 역할 분리로 유지보수 용이

## 3. 비동기 처리의 이해

API 호출은 **시간이 걸리는 작업**이므로 비동기로 처리해야 합니다.

### 동기 vs 비동기

```javascript
// ❌ 동기 방식 (React에서 불가능)
const data = fetch(url); // 응답을 기다리는 동안 화면이 멈춤
console.log(data);

// ✅ 비동기 방식
fetch(url)
  .then((response) => response.json())
  .then((data) => console.log(data)); // 응답이 오면 실행
console.log("이 코드가 먼저 실행됨");
```

### async/await (더 읽기 쉬운 비동기 코드)

```javascript
async function fetchData() {
  const response = await fetch(url); // 응답 대기
  const data = await response.json(); // JSON 변환 대기
  return data;
}
```

## 4. React의 생명주기와 API 호출

컴포넌트가 화면에 표시될 때 데이터를 가져오려면 **useEffect**를 사용합니다.

### 컴포넌트 생명주기

```
1. 마운트   → 컴포넌트가 화면에 처음 나타남
2. 업데이트 → 상태나 props가 변경됨
3. 언마운트 → 컴포넌트가 화면에서 사라짐
```

### useEffect로 API 호출

```javascript
import { useState, useEffect } from "react";

function UserList() {
  const [users, setUsers] = useState([]);

  useEffect(() => {
    // 컴포넌트가 마운트될 때 실행
    fetch("https://api.example.com/users")
      .then((res) => res.json())
      .then((data) => setUsers(data));
  }, []); // 빈 배열 = 한 번만 실행

  return (
    <ul>
      {users.map((user) => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}
```

## 5. 상태 관리의 3요소

API 연동 시 반드시 관리해야 할 세 가지 상태:

```javascript
const [data, setData] = useState(null); // 1. 데이터
const [loading, setLoading] = useState(true); // 2. 로딩 상태
const [error, setError] = useState(null); // 3. 에러 상태
```

### 상태 흐름도

```
초기 상태: loading=true, data=null, error=null
    ↓
API 요청 시작
    ↓
    ├─ 성공 → loading=false, data=응답데이터, error=null
    └─ 실패 → loading=false, data=null, error=에러메시지
```

### 완전한 예제

```javascript
function UserList() {
  const [users, setUsers] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const fetchUsers = async () => {
      try {
        setLoading(true);
        const response = await fetch("https://api.example.com/users");

        if (!response.ok) {
          throw new Error("서버 응답 실패");
        }

        const data = await response.json();
        setUsers(data);
        setError(null);
      } catch (err) {
        setError(err.message);
        setUsers(null);
      } finally {
        setLoading(false);
      }
    };

    fetchUsers();
  }, []);

  // 조건부 렌더링
  if (loading) return <div>로딩 중...</div>;
  if (error) return <div>에러: {error}</div>;
  if (!users) return <div>데이터 없음</div>;

  return (
    <ul>
      {users.map((user) => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}
```

## 6. HTTP 요청의 구조

### GET 요청 (조회)

```javascript
fetch("https://api.example.com/users/123")
  .then((response) => response.json())
  .then((data) => console.log(data));
```

### POST 요청 (생성)

```javascript
fetch("https://api.example.com/users", {
  method: "POST",
  headers: {
    "Content-Type": "application/json", // JSON 형식으로 보냄
  },
  body: JSON.stringify({
    // 객체를 JSON 문자열로 변환
    name: "홍길동",
    email: "hong@example.com",
  }),
})
  .then((response) => response.json())
  .then((data) => console.log("생성됨:", data));
```

### PUT 요청 (수정)

```javascript
fetch("https://api.example.com/users/123", {
  method: "PUT",
  headers: {
    "Content-Type": "application/json",
  },
  body: JSON.stringify({
    name: "홍길동 수정",
  }),
});
```

### DELETE 요청 (삭제)

```javascript
fetch("https://api.example.com/users/123", {
```

## 7. 상태 업데이트 패턴

### 생성 후 상태 업데이트

```javascript
const createUser = async (newUser) => {
  const response = await fetch("/api/users", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(newUser),
  });

  const createdUser = await response.json();

  // 기존 배열에 새 데이터 추가
  setUsers([...users, createdUser]);
};
```

### 수정 후 상태 업데이트

```javascript
const updateUser = async (id, updatedData) => {
  await fetch(`/api/users/${id}`, {
    method: "PUT",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(updatedData),
  });

  // 해당 ID의 데이터만 업데이트
  setUsers(
    users.map((user) => (user.id === id ? { ...user, ...updatedData } : user)),
  );
};
```

### 삭제 후 상태 업데이트

```javascript
const deleteUser = async (id) => {
  await fetch(`/api/users/${id}`, {
    method: "DELETE",
  });

  // 해당 ID 제외하고 필터링
  setUsers(users.filter((user) => user.id !== id));
};
```

## 8. 에러 처리

### 네트워크 에러 vs HTTP 에러

```javascript
try {
  const response = await fetch(url);

  // HTTP 상태 코드 확인 (404, 500 등)
  if (!response.ok) {
    throw new Error(`HTTP error! status: ${response.status}`);
  }

  const data = await response.json();
  setData(data);
} catch (error) {
  // 네트워크 에러 또는 파싱 에러
  setError(error.message);
}
```
