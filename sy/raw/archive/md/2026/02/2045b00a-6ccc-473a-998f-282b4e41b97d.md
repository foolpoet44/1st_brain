## 🎬 교육 영상 생성의 패러다임 전환

전통적인 교육 영상 제작은 "전문가의 영역"이었습니다. 촬영 장비, 편집 기술, 스튜디오 환경이 필요했죠. 하지만 AI는 이를 **"아이디어를 가진 사람이면 누구나"**의 영역으로 전환시키고 있습니다. 이는 단순한 기술적 진보가 아니라, **교육 콘텐츠의 민주화**라는 철학적 의미를 갖습니다.

## 💡 철학적 고찰: AI 교육 영상의 본질

AI가 교육 영상을 만든다고 할 때, 우리는 무엇을 잃고 무엇을 얻는가?

**잃는 것**: 인간 강사의 즉흥성, 학습자와의 실시간 상호작용, 예측 불가능한 통찰의 순간

**얻는 것**: 확장성, 일관성, 접근성, 개인화 가능성, 반복 학습의 용이성

교육의 본질은 "지식의 전달"이 아니라 **"변화의 촉진"**입니다. AI 영상은 지식 전달의 효율을 극대화하지만, 변화의 촉진은 여전히 인간 코치, 멘토, 퍼실리테이터의 몫입니다.

따라서 가장 이상적인 모델은 **블렌디드 러닝**: AI가 표준화된 지식을 전달하고, 인간이 맥락화된 적용을 돕는 구조입니다.

### 전체 워크플로우 코드

```python
"""
슬라이드→영상 자동 생성기
작성자: CSP (Creative Solution Provider)
철학: "Do it once, automate it forever"
"""

from pptx import Presentation
from pptx.util import Inches, Pt
from gtts import gTTS
from moviepy.editor import *
from PIL import Image
import os
from datetime import datetime

class EducationVideoAutomator:
    """교육 영상 자동 생성 클래스"""

    def __init__(self, output_dir="./output"):
        self.output_dir = output_dir
        os.makedirs(output_dir, exist_ok=True)

    def create_slide_deck(self, content_data):
        """
        Notion 데이터로 PPT 자동 생성

        content_data 구조:
        {
            'title': '리더십의 본질',
            'slides': [
                {
                    'title': '슬라이드 제목',
                    'content': '슬라이드 내용',
                    'narration': '나레이션 스크립트'
                },
                ...
            ]
        }
        """
        prs = Presentation()
        prs.slide_width = Inches(16)
        prs.slide_height = Inches(9)

        # 제목 슬라이드
        title_slide = prs.slides.add_slide(prs.slide_layouts[0])
        title_slide.shapes.title.text = content_data['title']
        subtitle = title_slide.placeholders[1]
        subtitle.text = f"생성일: {datetime.now().strftime('%Y-%m-%d')}\n제작: HR자동화시스템"

        # 내용 슬라이드들
        for slide_data in content_data['slides']:
            slide = prs.slides.add_slide(prs.slide_layouts[1])

            # 제목
            slide.shapes.title.text = slide_data['title']

            # 본문
            content_box = slide.placeholders[1]
            content_box.text = slide_data['content']

            # 폰트 설정 (한국어 가독성)
            for paragraph in content_box.text_frame.paragraphs:
                for run in paragraph.runs:
                    run.font.name = '맑은 고딕'
                    run.font.size = Pt(18)

        ppt_path = f"{self.output_dir}/{content_data['title']}.pptx"
        prs.save(ppt_path)
        print(f"✅ PPT 생성 완료: {ppt_path}")
        return ppt_path

    def generate_narrations(self, content_data):
        """
        각 슬라이드의 나레이션 음성 파일 생성
        """
        audio_files = []

        for idx, slide_data in enumerate(content_data['slides']):
            narration_text = slide_data.get('narration', slide_data['content'])

            # gTTS로 음성 생성 (한국어)
            tts = gTTS(text=narration_text, lang='ko', slow=False)

            audio_path = f"{self.output_dir}/narration_{idx}.mp3"
            tts.save(audio_path)
            audio_files.append(audio_path)

            print(f"🎙️ 나레이션 생성: 슬라이드 {idx+1}")

        return audio_files

    def ppt_to_images(self, ppt_path):
        """
        PPT를 이미지로 변환
        (실제로는 LibreOffice 커맨드라인 사용)
        """
        # LibreOffice headless 모드로 PPT → PDF → 이미지
        pdf_path = ppt_path.replace('.pptx', '.pdf')

        # PPT → PDF 변환
        os.system(f'libreoffice --headless --convert-to pdf --outdir {self.output_dir} {ppt_path}')

        # PDF → 이미지 변환 (ImageMagick 사용)
        image_dir = f"{self.output_dir}/slides"
        os.makedirs(image_dir, exist_ok=True)
        os.system(f'convert -density 300 {pdf_path} {image_dir}/slide_%03d.png')

        # 이미지 파일 목록 반환
        images = sorted([f"{image_dir}/{f}" for f in os.listdir(image_dir) if f.endswith('.png')])
        print(f"🖼️ 슬라이드 이미지 생성: {len(images)}개")
        return images

    def create_video(self, image_files, audio_files, content_data):
        """
        이미지 + 음성 → 영상 합성
        """
        clips = []

        for idx, (image_path, audio_path) in enumerate(zip(image_files, audio_files)):
            # 음성 길이 계산
            audio_clip = AudioFileClip(audio_path)
            duration = audio_clip.duration

            # 이미지 클립 생성
            image_clip = ImageClip(image_path).set_duration(duration)

            # 음성 추가
            video_clip = image_clip.set_audio(audio_clip)

            clips.append(video_clip)
            print(f"🎬 슬라이드 {idx+1} 처리 완료 ({duration:.1f}초)")

        # 모든 클립 연결
        final_video = concatenate_videoclips(clips, method="compose")

        # 인트로/아웃트로 추가 (선택사항)
        # final_video = self.add_intro_outro(final_video, content_data)

        # 영상 저장
        video_path = f"{self.output_dir}/{content_data['title']}_교육영상.mp4"
        final_video.write_videofile(
            video_path,
            fps=24,
            codec='libx264',
            audio_codec='aac',
            temp_audiofile='temp-audio.m4a',
            remove_temp=True
        )

        print(f"✅ 최종 영상 생성 완료: {video_path}")
        return video_path

    def run(self, content_data):
        """
        전체 파이프라인 실행
        """
        print("🚀 교육 영상 자동 생성 시작...")

        # 1. PPT 생성
        ppt_path = self.create_slide_deck(content_data)

        # 2. 나레이션 생성
        audio_files = self.generate_narrations(content_data)

        # 3. PPT → 이미지
        image_files = self.ppt_to_images(ppt_path)

        # 4. 영상 합성
        video_path = self.create_video(image_files, audio_files, content_data)

        print(f"""
        ✨ 완료! ✨

        생성된 파일:
        - PPT: {ppt_path}
        - 영상: {video_path}

        총 소요 시간: 약 3-5분
        CSP님이 한 일: 0번 클릭
        """)

        return video_path


# 실행 예제
if __name__ == "__main__":

    # Notion에서 가져온 데이터 (나중에 API 연동)
```

## 🎯 에이전트의 철학적 설계

### 에이전트 vs 스크립트의 차이

**일반 스크립트**:

```
입력 → 처리 → 출력
(단순 변환)
```

**전문 에이전트**:

```
입력 → 이해 → 판단 → 최적화 → 실행 → 학습
(지능형 의사결정)
```

### CSP님의 에이전트가 해야 할 일

```
1. 교육학적 분석
   - 이 PPT는 어떤 유형의 교육인가? (개념/절차/태도)
   - 학습 목표는 무엇인가?
   - 최적 교수법은?

2. 콘텐츠 최적화
   - 슬라이드가 너무 많으면 → 자동 통합
   - 텍스트가 너무 많으면 → 핵심만 추출
   - 내레이션 스크립트 자동 생성

3. 감정/톤 매칭
   - 도입부 → 밝고 친근하게
   - 핵심 개념 → 진지하고 명확하게
   - 사례 → 대화하듯 자연스럽게

4. 품질 검증
   - 음성 품질 체크
   - 영상 길이 최적화 (7-15분)
   - 자막 자동 생성

5. 학습 효과 예측
   - 이 영상의 예상 학습 효과는?
   - 개선 포인트는?
```

```python
Education Video Generation Agent

철학: "AI가 교수설계자처럼 사고한다"
작성자: CSP

import anthropic
import os
from pathlib import Path
import json
from typing import Dict, List, Any
from dataclasses import dataclass
from enum import Enum

class LearningType(Enum):
    """학습 유형 분류"""
    CONCEPTUAL = "개념학습"      # 리더십이란?
    PROCEDURAL = "절차학습"      # 피드백 방법
    ATTITUDINAL = "태도학습"     # 리더의 마음가짐
    CASE_STUDY = "사례학습"      # 실제 사례 분석

class InstructionalStrategy(Enum):
    """교수 전략"""
    EXPOSITORY = "설명형"        # 직접 설명
    INQUIRY = "탐구형"          # 질문과 발견
    DEMONSTRATION = "시연형"     # 보여주기
    DISCUSSION = "토론형"       # 대화식

@dataclass
class EducationalAnalysis:
    """교육학적 분석 결과"""
    learning_type: LearningType
    learning_objectives: List[str]
    target_audience: str
    optimal_strategy: InstructionalStrategy
    estimated_duration: int  # 분
    cognitive_load: str  # low, medium, high
    recommendations: List[str]

@dataclass
class NarrationScript:
    """나레이션 스크립트"""
    slide_number: int
    text: str
    tone: str  # friendly, professional, serious, conversational
    emphasis_words: List[str]
    pause_points: List[int]  # 쉼표 위치
    estimated_duration: float  # 초

class EducationVideoAgent:
    """교육 영상 생성 전문 에이전트"""

    def __init__(self):
        self.client = anthropic.Anthropic(
            api_key=os.environ.get("ANTHROPIC_API_KEY")
        )
        self.knowledge_base = self._load_knowledge_base()

    def _load_knowledge_base(self) -> Dict:
        """교육학 지식 베이스 로드"""
        return {
            "instructional_design": {
                "ADDIE": "Analysis, Design, Development, Implementation, Evaluation",
                "Bloom_Taxonomy": ["Remember", "Understand", "Apply", "Analyze", "Evaluate", "Create"],
                "Cognitive_Load": "Manage intrinsic, extraneous, and germane load",
                "Multimedia_Principles": [
                    "Coherence: 불필요한 정보 제거",
                    "Signaling: 핵심 내용 강조",
                    "Redundancy: 중복 최소화",
                    "Spatial_Contiguity: 관련 정보 근접 배치",
                    "Temporal_Contiguity: 동시 제시",
                    "Segmenting: 작은 단위로 분할",
                    "Modality: 시각+청각 채널 활용",
                    "Personalization: 대화체 사용"
                ]
            },
            "narration_guidelines": {
                "optimal_pace": "150-160 단어/분",
                "sentence_length": "15-20단어",
                "pause_duration": "1-2초",
                "tone_matching": {
                    "introduction": "warm, welcoming",
                    "concept": "clear, authoritative",
                    "example": "conversational, relatable",
                    "conclusion": "motivating, action-oriented"
                }
            }
        }

    def analyze_ppt(self, ppt_path: str) -> EducationalAnalysis:
        """
        PPT 교육학적 분석
        Claude가 교수설계자처럼 분석
        """

        # 1. PPT 내용 추출
        content = self._extract_ppt_content(ppt_path)

        # 2. Claude에게 교육학적 분석 요청
        analysis_prompt = f"""
당신은 17년 경력의 HR 교육 전문가이자 교수설계자입니다.
다음 PPT 내용을 분석하여 교육학적 통찰을 제공하세요.

<ppt_content>
{json.dumps(content, ensure_ascii=False, indent=2)}
</ppt_content>

<analysis_framework>
1. 학습 유형 분류:
   - CONCEPTUAL: 개념/이론 설명
   - PROCEDURAL: 절차/방법 안내
   - ATTITUDINAL: 태도/가치관 형성
   - CASE_STUDY: 사례 분석

2. 학습 목표 추출:
   - Bloom의 분류학 기준
   - 구체적이고 측정 가능하게

3. 대상 학습자:
   - 직급, 경력, 사전 지식

4. 최적 교수 전략:
   - EXPOSITORY: 직접 설명
   - INQUIRY: 질문과 탐구
   - DEMONSTRATION: 시연
   - DISCUSSION: 대화식

5. 인지 부하 평가:
   - low: 단순하고 명확
   - medium: 적절한 복잡도
   - high: 많은 정보, 복잡한 개념

6. 개선 권장사항:
   - Mayer의 멀티미디어 학습 원리 적용
</analysis_framework>

JSON 형식으로 응답하세요:

  "optimal_strategy": "EXPOSITORY|INQUIRY|DEMONSTRATION|DISCUSSION",

  "cognitive_load": "low|medium|high",
  "recommendations": ["권장사항1", "권장사항2", ...]
}}
"""

        response = self.client.messages.create(
            model="claude-sonnet-4-20250514",
            max_tokens=4096,
            messages=[{
                "role": "user",
                "content": analysis_prompt
            }]
        )

        # JSON 파싱
        analysis_json = json.loads(response.content[0].text)

        # 분석 결과 객체화
        analysis = EducationalAnalysis(
            learning_type=LearningType[analysis_json["learning_type"]],
            learning_objectives=analysis_json["learning_objectives"],
            target_audience=analysis_json["target_audience"],
            optimal_strategy=InstructionalStrategy[analysis_json["optimal_strategy"]],
            estimated_duration=analysis_json["estimated_duration"],
            cognitive_load=analysis_json["cognitive_load"],
            recommendations=analysis_json["recommendations"]
        )

        print(f"""
╔══════════════════════════════════════════╗
║   📊 교육학적 분석 완료                    ║

🎯 학습 유형: {analysis.learning_type.value}
📌 학습 목표:
{chr(10).join(f'   • {obj}' for obj in analysis.learning_objectives)}

👥 대상 학습자: {analysis.target_audience}
📚 추천 교수법: {analysis.optimal_strategy.value}
⏱️  예상 시간: {analysis.estimated_duration}분
🧠 인지 부하: {analysis.cognitive_load}

💡 개선 권장사항:
{chr(10).join(f'   • {rec}' for rec in analysis.recommendations)}
        """)

        return analysis

    def generate_narration_scripts(
        self,
        content: Dict,
        analysis: EducationalAnalysis
    ) -> List[NarrationScript]:
        """
        교육학적 분석 기반 최적 나레이션 스크립트 생성
        """

        scripts = []

        for idx, slide in enumerate(content['slides']):

            # 슬라이드 위치에 따른 톤 결정
            if idx == 0:
                section = "introduction"
            elif idx == len(content['slides']) - 1:
                section = "conclusion"
            else:
                section = "concept" if analysis.learning_type == LearningType.CONCEPTUAL else "example"

            tone = self.knowledge_base["narration_guidelines"]["tone_matching"][section]

            # Claude에게 나레이션 스크립트 생성 요청
            script_prompt = f"""
당신은 전문 교육 나레이터입니다.
다음 슬라이드 내용을 자연스러운 음성 스크립트로 변환하세요.

<slide_content>
제목: {slide['title']}
내용: {slide['content']}
</slide_content>

<context>
- 학습 유형: {analysis.learning_type.value}
- 교수 전략: {analysis.optimal_strategy.value}
- 섹션: {section}
- 톤: {tone}
- 대상: {analysis.target_audience}
</context>

<guidelines>
1. 대화체로 작성 (존댓말 사용)
2. 한 문장 15-20단어
3. 전문 용어는 쉽게 풀어서 설명
4. 강조할 단어는 <em>태그로 표시
5. 쉼표(pause)가 필요한 곳은 |로 표시
6. 자연스러운 호흡과 리듬

예시:
"리더십은 | <em>영향력</em>입니다. | 타인의 자발적 행동을 | 이끌어내는 능력이죠."
</guidelines>

스크립트만 출력하세요 (추가 설명 불필요):
"""

            response = self.client.messages.create(
                model="claude-sonnet-4-20250514",
                max_tokens=1024,
                messages=[{"role": "user", "content": script_prompt}]
            )

            raw_script = response.content[0].text.strip()

            # 강조 단어 추출
            import re
            emphasis_words = re.findall(r'<em>(.*?)</em>', raw_script)
            clean_script = re.sub(r'<em>|</em>', '', raw_script)

            # pause 위치 추출
            pause_points = [i for i, char in enumerate(clean_script) if char == '|']
            clean_script = clean_script.replace('|', '')

            # 예상 시간 계산 (150단어/분)
            word_count = len(clean_script.split())
            duration = (word_count / 150) * 60  # 초

            script = NarrationScript(
                slide_number=idx + 1,
                text=clean_script,
                tone=tone,
                emphasis_words=emphasis_words,
                pause_points=pause_points,
                estimated_duration=duration
            )

            scripts.append(script)

            print(f"✍️  슬라이드 {idx+1} 스크립트 생성 ({duration:.1f}초)")

        return scripts

    def _extract_ppt_content(self, ppt_path: str) -> Dict:
        """PPT 내용 추출 (간단 버전)"""
        from pptx import Presentation

        prs = Presentation(ppt_path)
        content = {
            'title': prs.slides[0].shapes.title.text if prs.slides else "Untitled",
            'slides': []
        }

        for slide in prs.slides:
            slide_data = {
                'title': slide.shapes.title.text if slide.shapes.title else "",
                'content': self._extract_text_from_shapes(slide.shapes)
            }
            content['slides'].append(slide_data)

        return content

    def _extract_text_from_shapes(self, shapes):
        """슬라이드에서 텍스트 추출"""
        text_parts = []
        for shape in shapes:
            if hasattr(shape, "text"):
                text_parts.append(shape.text)
        return "\n".join(text_parts)
```
